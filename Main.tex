\documentclass[10pt,a4paper,twoside]{class/main}

%----------------------------------------------------------
% 基本宏包配置
%----------------------------------------------------------
\usepackage[english]{babel}
\usepackage{graphicx}
\usepackage{booktabs}
\usepackage[table]{xcolor}

%----------------------------------------------------------
% 主题色配置 (可选: red, purple, blue)
%----------------------------------------------------------
\academictheme{red}  % 切换主题色：red / purple / blue

%----------------------------------------------------------
% 论文信息配置 (Metadata)
%----------------------------------------------------------

\title{Your Paper Title}

% 致谢/资助/版权信息（显示在首页底部）
% \license{This work was supported by ...}

% 页眉/页脚信息
\leadauthor{Author One et al.}
\institution{University Name}

% 开启功能模块
\setabstractenabled{true}
\setbool{corres-info}{true}

%----------------------------------------------------------

%----------------------------------------------------------
\begin{abstract}

\begin{document}

\maketitle

\dropcapstart{T}
\section{Introduction}
Introduce the problem, motivation, and core contribution. This section should explain why the problem matters and summarize the paper's main idea.

\section{Related Work}
Summarize the most relevant literature and clearly position your work against existing methods.

\section{Method}
Describe your model, algorithm, system design, or theoretical framework. Use figures, equations, and tables as needed.

\section{Experiments}
Report datasets, experimental settings, baselines, metrics, and the main quantitative/qualitative results.

\section{Discussion}
Analyze limitations, assumptions, and broader implications where appropriate.

\section{Conclusion}
Conclude with the main takeaway, practical impact, and possible future work.
